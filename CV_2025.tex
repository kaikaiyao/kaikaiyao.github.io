\section*{Research Experience}
\cvrole{Smudged Fingerprints: Robustness of AI Image Fingerprinting}{2024 -- 2025}{University of Edinburgh}
\begin{itemize}
    \item Conducted the first systematic security evaluation of model fingerprint detection for generative image attribution, formalizing white-box and black-box threat models and implementing five removal and forgery attack strategies across RGB, frequency, and learned-feature domains on 12 state-of-the-art GAN, VAE, and diffusion models.
    \item Identified severe robustness gaps, utility-robustness trade-offs, and fundamental limitations of passive fingerprinting for adversarial provenance.
\end{itemize}

\cvrole{Fingerprinting Generative Models Against Malicious Providers}{2024 -- 2025}{University of Edinburgh}
\begin{itemize}
    \item Developed robust fingerprint extraction methods for GANs and diffusion models under threat models where the provider itself may be adversarial.
    \item Built a comprehensive evaluation framework across models and attack scenarios, supporting research on AI security and model provenance.
\end{itemize}

\newpage
\cvrole{Differential Privacy's Disparate Impact in Machine Learning}{2023 -- 2024}{University of Edinburgh}
\begin{itemize}
    \item Conducted a systematic fairness analysis of differentially private machine learning, developing a taxonomy of disparity-inducing factors across privacy mechanisms, model architectures, and data distributions.
    \item Used causal and empirical analysis to identify dataset size and group distance to decision boundaries as key drivers of DP-induced unfairness, and evaluated mitigation strategies.
\end{itemize}

\cvrole{Deep Learning for Single-Cell Morphology and Measurement}{2018 -- 2020}{Johns Hopkins University}
\begin{itemize}
    \item Developed label-free deep learning pipelines for high-throughput biological image analysis, including U-Net-based morphology prediction, cell-type classification, automated preprocessing, and unsupervised morphological phenotyping.
    \item Produced multiple peer-reviewed outputs from interdisciplinary AI research in computational biology and scientific imaging.
\end{itemize}
